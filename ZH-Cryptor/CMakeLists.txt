cmake_minimum_required(VERSION 3.20)
project(ZHCryptor VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "ZH Cryptor is Windows-only.")
endif()
add_executable(ZHCryptor WIN32 src/main.cpp)
target_compile_features(ZHCryptor PRIVATE cxx_std_17)
target_compile_definitions(ZHCryptor PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_compile_options(ZHCryptor PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(ZHCryptor PRIVATE bcrypt comctl32 comdlg32 shell32 advapi32 ole32)
set_property(TARGET ZHCryptor PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
